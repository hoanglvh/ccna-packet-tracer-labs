 remark Guest VLAN 20 - Internet only
 permit udp any any eq bootps

 deny ip 192.168.20.0 0.0.0.255 192.168.0.0 0.0.255.255
 permit tcp 192.168.20.0 0.0.0.255 any eq www
 permit tcp 192.168.20.0 0.0.0.255 any eq 443
 permit icmp 192.168.20.0 0.0.0.255 any echo
 exit
ip access-list extended SERVER-PROTECT
 permit tcp any host 192.168.30.10 eq www
 permit tcp any host 192.168.30.10 eq 443
 permit icmp 192.168.10.0 0.0.0.255 host 192.168.30.10 echo
 permit ip 192.168.99.0 0.0.0.255 192.168.30.0 0.0.0.255
 deny ip any any
 exit
access-list 1 remark Inside hosts allowed to use PAT
access-list 1 permit 192.168.10.0 0.0.0.255
access-list 1 permit 192.168.20.0 0.0.0.255
